DECLARE @i INT = 61;
DECLARE @max INT = 90;

WHILE @i <= @max
BEGIN
    INSERT INTO [dbo].[tb_UserInfo] (
      [UId], [UserNo], [UserPwd], [UserName], [UserType], [UserSchoolId], [UserClassId], 
      [StudentNo], [UserSex], [UserYear], [UserIdentity], [UserPhone], [UserEmail], 
      [UserQQ], [UserWX], [UserPic], [State], [Operator], [AddOperator], [AddTime], 
      [Custom1], [Custom2], [Custom3]
    )
    VALUES (
      NEWID(), -- UId (uniqueidentifier)
      @i, -- UserNo (int)
      'password', -- UserPwd (varchar)
      'User ' + CAST(@i AS VARCHAR), -- UserName (varchar)
      'Type', -- UserType (varchar)
      'SchoolId', -- UserSchoolId (varchar)
      'ClassId', -- UserClassId (varchar)
      @i, -- StudentNo (int)
      'Sex', -- UserSex (varchar)
      'Year', -- UserYear (varchar)
      'Identity', -- UserIdentity (varchar)
      'Phone', -- UserPhone (varchar)
      'Email', -- UserEmail (varchar)
      'QQ', -- UserQQ (varchar)
      'WX', -- UserWX (varchar)
      'Pic', -- UserPic (varchar)
      'State', -- State (varchar)
      'Operator', -- Operator (varchar)
      'AddOperator', -- AddOperator (varchar)
      GETDATE(), -- AddTime (datetime)
      'Custom1', -- Custom1 (varchar)
      'Custom2', -- Custom2 (varchar)
      'Custom3' -- Custom3 (varchar)
    );

    SET @i = @i + 1;
END
