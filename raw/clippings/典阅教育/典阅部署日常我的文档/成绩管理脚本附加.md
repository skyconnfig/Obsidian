CREATE VIEW [dbo].[V_AllTeamScore]
AS
SELECT     dbo.tb_ExaminationResult.ER_Type, dbo.tb_ExaminationResult.ER_Score, dbo.tb_ExaminationResult.ER_State, dbo.tb_ExaminationResult.ER_AddTime, dbo.tb_Team.TeamName, 
                      dbo.tb_Team.TeacherId, tb_UserInfo_1.UserName AS teacName, dbo.tb_UserInfo.UserName AS stuName, dbo.tb_Team.Id AS teamId
FROM         dbo.tb_ExaminationResult INNER JOIN
                      dbo.tb_UserInfo ON dbo.tb_ExaminationResult.ER_MId = dbo.tb_UserInfo.UId INNER JOIN
                      dbo.tb_Team ON dbo.tb_UserInfo.UserClassId = dbo.tb_Team.Id INNER JOIN
                      dbo.tb_UserInfo AS tb_UserInfo_1 ON dbo.tb_Team.TeacherId = tb_UserInfo_1.UId

GO
