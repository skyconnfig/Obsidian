```python
-- 生成6位随机数并进行MD5加密
WITH RandomCTE AS (
    SELECT 
        ABS(CHECKSUM(NEWID())) % 900000 + 100000 AS RandomNumber
)
SELECT 
    RandomNumber AS [随机6位数],
    CONVERT(VARCHAR(32), HASHBYTES('MD5', CAST(RandomNumber AS VARCHAR(6))), 2) AS [MD5加密结果]
FROM RandomCTE;
```

### 理财规划 如果是 =888888 则生成六位数据密码
```python
--理财规划随机密码
UPDATE [dbo].[bi_BankTeller] 
SET [Password] = RIGHT('000000' + CAST(ABS(CHECKSUM(NEWID())) % 1000000 as VARCHAR(6)), 6)
WHERE Student_No LIKE '202500%'
```
