# 通过注册码更新日期到站点名称以及V2版本

```powershell
# 导入模组
Import-Module WebAdministration


# 定义连接字符串（请根据实际情况修改）  
$connectionString = "Server=120.77.10.122;Database=bug;User Id=sa;Password=Dysoft0706!@#;"

# 创建 SqlConnection 对象  
$connection = New-Object System.Data.SqlClient.SqlConnection($connectionString)

# 获取所有 IIS 站点
$sites = Get-Website

try {
    # 打开连接  
    $connection.Open()

    foreach ($site in $sites) {
        $originalName = $site.name
        $sitePath = (Get-ItemProperty "IIS:\Sites\$originalName").PhysicalPath
        $webConfigPath = "$sitePath\web.config"

        # 检查 web.config 文件是否存在
        if (Test-Path $webConfigPath) {
            [xml]$config = Get-Content $webConfigPath

            # 检查 web.config 中是否存在 key="SerialNo" 的元素
            $serialNoElement = $config.configuration.appSettings.add | Where-Object { $_.key -eq "SerialNo" }

            if ($serialNoElement) {
                $serialNoValue = $serialNoElement.value

                # 去掉 yyyy-mm-dd 格式的日期
                if ($originalName -match '\d{4}-\d{2}-\d{2}$') {
                    $newName = $originalName -replace '\d{4}-\d{2}-\d{2}$', ''
                }
                # 去掉 yyyymmdd 格式的日期
                elseif ($originalName -match '\d{8}$') {
                    $newName = $originalName -replace '\d{8}$', ''
                }
                else {
                    $newName = $originalName
                }

                # 创建 SqlCommand 对象  
                $sqlQuery = "SELECT TOP 1 CONVERT(varchar, endtime, 112) AS endtime FROM dbo.tb_TimeMD5 WHERE miyao = '$serialNoValue'"
                $sqlCommand = New-Object System.Data.SqlClient.SqlCommand($sqlQuery, $connection)

                # 执行查询并获取 SqlDataReader 对象  
                $reader = $sqlCommand.ExecuteReader()

                # 重命名站点
                if ($reader.Read()) {
                    $endtime = $reader["endtime"].ToString()
                    $newSiteName = "$newName$endtime"
                    Set-ItemProperty "IIS:\Sites\$originalName" -Name name -Value $newSiteName
                    Write-Host "$serialNoValue`t$newSiteName"
                }

                # 关闭 SqlDataReader 对象
                $reader.Close()
            }
        }
    }
}
catch {
    # 处理任何异常  
    Write-Host "Error: " $_.Exception.Message
}
finally {
    # 确保关闭连接  
    if ($connection.State -eq "Open") {
        $connection.Close()
    }
}
```

```plain
# 1. 临时设置脚本执行策略（解决脚本无法执行问题，执行完脚本后可根据需要改回原策略）
Set-ExecutionPolicy Bypass -Scope Process -Force

try {
    # 2. 导入 WebAdministration 模块，添加异常捕获确保模块加载反馈清晰
    if (-not (Get-Module -Name WebAdministration -ErrorAction SilentlyContinue)) {
        Import-Module WebAdministration -ErrorAction Stop
        Write-Host "已成功加载 WebAdministration 模块"
    }

    # 3. 定义连接字符串（请根据实际情况修改）  
    $connectionString = "Server=120.77.10.122;Database=bug;User Id=sa;Password=Dysoft0706!@#;"
    $connection = New-Object System.Data.SqlClient.SqlConnection($connectionString)
    $connection.Open()

    # 4. 输出表头，用更规范的格式和分隔符
    Write-Host "站点名称`t状态`t序列号`t过期日期`t许可证状态"
    Write-Host ("-" * 60)

    # 5. 准确获取所有 IIS 站点（兼容不同环境，优先用 Get-Website 补充 Get-WebSite ）
    $sites = @()
    if (Get-Command Get-WebSite -ErrorAction SilentlyContinue) {
        $sites = Get-WebSite
    } else {
        $sites = Get-Website
    }

    foreach ($site in $sites) {
        $siteName = $site.Name
        $siteState = $site.State.ToString()

        # 6. 根据站点状态设置显示颜色
        $siteStateColor = switch ($siteState) {
            "Started" { "Green" }
            default { "Yellow" }
        }

        $sitePath = (Get-ItemProperty "IIS:\Sites\$siteName" -ErrorAction SilentlyContinue).PhysicalPath
        $webConfigPath = if ($sitePath) { Join-Path $sitePath "web.config" } else { "" }

        if (Test-Path $webConfigPath -PathType Leaf) {
            try {
                [xml]$config = Get-Content $webConfigPath -ErrorAction Stop
                $serialNoElement = $config.configuration.appSettings.add | 
                    Where-Object { $_.key -eq "SerialNo" } -ErrorAction SilentlyContinue

                if ($serialNoElement) {
                    $serialNoValue = $serialNoElement.value
                    $sqlQuery = "SELECT TOP 1 
                                CONVERT(varchar, endtime, 23) AS formattedEndtime,
                                (CASE WHEN endtime < GETDATE() THEN '已过期' ELSE '有效' END) AS licenseStatus
                             FROM dbo.tb_TimeMD5 
                             WHERE miyao = @SerialNo"
                    
                    $sqlCommand = $connection.CreateCommand()
                    $sqlCommand.CommandText = $sqlQuery
                    $sqlParam = $sqlCommand.Parameters.Add("@SerialNo", [System.Data.SqlDbType]::VarChar)
                    $sqlParam.Value = $serialNoValue

                    $reader = $sqlCommand.ExecuteReader()
                    if ($reader.Read()) {
                        $endtime = $reader["formattedEndtime"].ToString()
                        $licenseStatus = $reader["licenseStatus"].ToString()
                        
                        $licenseColor = if ($licenseStatus -eq "已过期") { "Red" } else { "Green" }

                        Write-Host "$siteName`t" -NoNewline
                        Write-Host "$siteState`t" -ForegroundColor $siteStateColor -NoNewline
                        Write-Host "$serialNoValue`t$endtime`t" -NoNewline
                        Write-Host "$licenseStatus" -ForegroundColor $licenseColor
                    } else {
                        Write-Host "$siteName`t" -NoNewline
                        Write-Host "$siteState`t" -ForegroundColor $siteStateColor -NoNewline
                        Write-Host "$serialNoValue`t未找到匹配记录" -ForegroundColor Yellow
                    }
                    $reader.Close()
                } else {
                    Write-Host "$siteName`t" -NoNewline
                    Write-Host "$siteState`t" -ForegroundColor $siteStateColor -NoNewline
                    Write-Host "未找到序列号配置" -ForegroundColor Gray
                }
            } catch {
                Write-Host "$siteName`t" -NoNewline
                Write-Host "$siteState`t" -ForegroundColor $siteStateColor -NoNewline
                Write-Host "读取 web.config 失败: $_" -ForegroundColor Red
            }
        } else {
            Write-Host "$siteName`t" -NoNewline
            Write-Host "$siteState`t" -ForegroundColor $siteStateColor -NoNewline
            Write-Host "未找到 web.config 文件" -ForegroundColor Gray
        }
    }
} catch {
    Write-Host "执行过程中发生错误: $_" -ForegroundColor Red
} finally {
    if ($connection -and $connection.State -eq "Open") {
        $connection.Close()
    }
    # 7. （可选）执行完脚本后恢复执行策略到原来的状态，若要保留 Bypass 可删除以下代码
    $originalPolicy = Get-ExecutionPolicy -Scope Process
    Set-ExecutionPolicy $originalPolicy -Scope Process -Force
}
```
