# 批量还原数据库

```sql
--创建过程

CREATE PROCEDURE [dbo].[sp_Restore_DB]
@LocalBackupPath AS VARCHAR(255)='C:\',
@name SYSNAME='dbname'
 AS

DECLARE @CommandString AS VARCHAR(8000)
DECLARE @SQL AS VARCHAR(8000)
DECLARE @LogicalName_Data VARCHAR(50),@LogicalName_Log VARCHAR(50)
DECLARE @PhysicalName_Data VARCHAR(50),@PhysicalName_Log VARCHAR(50)
DECLARE @pass AS VARCHAR(10),@login AS SYSNAME ,@db SYSNAME
DECLARE @sql_db VARCHAR(100),@sql1 VARCHAR(100),@sql2 VARCHAR(100)
DECLARE @sql3 VARCHAR(100)


CREATE TABLE #stage
(
 LogicalName VARCHAR(50),
 PhysicalName VARCHAR(255),
 Type CHAR(1),
 FileGroupName  VARCHAR(50),
 [Size]VARCHAR(50),
 [MaxSize]VARCHAR(50)
)

----标识数据库的逻辑和物理名称文件名
INSERT INTO #stage EXEC('RESTORE FILELISTONLY FROM DISK=N''' + @LocalBackupPath+ '\'+  'dbname.bak''')

SELECT @LogicalName_Data=LogicalName ,
@PhysicalName_Data=RIGHT(PhysicalName, CHARINDEX('\', REVERSE(PhysicalName))-1),
@PhysicalName_Log=(SELECT RIGHT(PhysicalName, CHARINDEX('\', REVERSE(PhysicalName))-1)  FROM #stage WHERE Type='L'),
@LogicalName_Log=(SELECT LogicalName  FROM #stage WHERE Type='L')
FROM #stage WHERE Type='D'

---为数据库要存储的文件
SET @CommandString = 'IF NOT EXIST "' + @LocalBackupPath + '\' + @name + '" MD "' + @LocalBackupPath + '\' + @name + '"'
EXEC master.dbo.xp_cmdshell @CommandString

----执行恢复操作
---SET  @SQL = 'RESTORE DATABASE [' + @name + '] FROM DISK = N''' + @LocalBackupPath +  @name +   '.bak''
SET  @SQL = 'RESTORE DATABASE [' + @name + '] FROM DISK = N''' + @LocalBackupPath +  'dbname.bak''
WITH RECOVERY,
   MOVE '''+@LogicalName_Data+''' TO  ''' + @LocalBackupPath + @name + '\' +@PhysicalName_Data+''',
   MOVE '''+@LogicalName_Log+ ''' TO  ''' + @LocalBackupPath + @name + '\' +@PhysicalName_Log+''''

EXEC (@SQL)


--登录并授予他们访问数据库的权限
SET @login='loginname'
SET @pass='pass'
SET @db=@name
   IF NOT EXISTS (SELECT * FROM master.dbo.syslogins WHERE loginname=@login)
   BEGIN
   EXEC sp_addlogin @loginame = @login, @passwd =@pass, @defdb = @db
   END

SET @sql_db='USE '+@name
SET @sql1=(' IF NOT EXISTS (SELECT * FROM sysusers WHERE name= '''+@login+''')')
SET @sql2=(' BEGIN EXEC sp_grantdbaccess '''+@login+'''')
SET @sql3=('  EXEC sp_addrolemember ''db_owner'','''+@login+''' END')
EXEC (@sql_db+@sql1+@sql2+@sql3)
-----Adding the dbname user to the msdb database to run jobs
SET @name='msdb'
SET @sql_db='USE '+@name
SET @sql1=(' IF NOT EXISTS (SELECT * FROM sysusers WHERE name= '''+@login+''')')
SET @sql2=(' BEGIN EXEC sp_grantdbaccess '''+@login+'''')
SET @sql3=('  EXEC sp_addrolemember ''db_owner'','''+@login+''' END')
EXEC (@sql_db+@sql1+@sql2+@sql3)

GO
```
