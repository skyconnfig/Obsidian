# 全部数据库清理和数据库收缩，定义数据库清理

### 数据库收缩（紧急情况）
```python
-- 方式1：收缩所有数据库日志文件（适用于紧急释放空间）
DECLARE @dbName NVARCHAR(128)
DECLARE @logFile NVARCHAR(128)
DECLARE @sql NVARCHAR(MAX)

DECLARE db_cursor CURSOR FOR 
SELECT name 
FROM sys.databases 
WHERE database_id > 4 -- 排除系统数据库
AND state = 0 -- 仅在线数据库

OPEN db_cursor  
FETCH NEXT FROM db_cursor INTO @dbName  

WHILE @@FETCH_STATUS = 0  
BEGIN  
    -- 获取日志文件名
    SET @logFile = (
        SELECT TOP 1 name 
        FROM sys.master_files 
        WHERE database_id = DB_ID(@dbName) AND type = 1
    )

    -- 生成动态SQL
    SET @sql = N'
    USE [' + @dbName + '];
    ALTER DATABASE [' + @dbName + '] SET RECOVERY SIMPLE;
    DBCC SHRINKFILE (N''' + @logFile + ''' , 0, TRUNCATEONLY);
    ALTER DATABASE [' + @dbName + '] SET RECOVERY FULL;'

    -- 执行命令
    BEGIN TRY
        EXEC sp_executesql @sql
        PRINT '成功处理数据库: ' + @dbName
    END TRY
    BEGIN CATCH
        PRINT '处理失败: ' + @dbName + ' | 错误: ' + ERROR_MESSAGE()
    END CATCH

    FETCH NEXT FROM db_cursor INTO @dbName  
END  

CLOSE db_cursor  
DEALLOCATE db_cursor
```





### -- 方式2：推荐的标准维护方法（备份+收缩）
```python
-- 方式2：推荐的标准维护方法（备份+收缩）
DECLARE @dbName NVARCHAR(128)
DECLARE @logFile NVARCHAR(128)
DECLARE @backupPath NVARCHAR(500) = 'D:\LogBackup\' -- 需修改为实际路径

DECLARE db_cursor CURSOR FOR 
SELECT name 
FROM sys.databases 
WHERE database_id > 4 
AND recovery_model IN (1,2) -- 仅处理FULL/BULK_LOGGED
AND state = 0

OPEN db_cursor  
FETCH NEXT FROM db_cursor INTO @dbName  

WHILE @@FETCH_STATUS = 0  
BEGIN  
    -- 备份事务日志
    SET @sql = N'BACKUP LOG [' + @dbName + '] TO DISK = ''' 
              + @backupPath + @dbName + '_LogBackup_' 
              + CONVERT(NVARCHAR(20), GETDATE(), 112) + '.trn'' WITH INIT'
    BEGIN TRY
        EXEC sp_executesql @sql
    END TRY
    BEGIN CATCH
        PRINT '日志备份失败: ' + @dbName + ' | 错误: ' + ERROR_MESSAGE()
    END CATCH

    -- 收缩日志文件
    SET @logFile = (
        SELECT TOP 1 name 
        FROM sys.master_files 
        WHERE database_id = DB_ID(@dbName) AND type = 1
    )
    
    SET @sql = N'USE [' + @dbName + '];
                DBCC SHRINKFILE (N''' + @logFile + ''', 1024)' -- 收缩到1GB
    
    BEGIN TRY
        EXEC sp_executesql @sql
        PRINT '成功处理数据库: ' + @dbName
    END TRY
    BEGIN CATCH
        PRINT '收缩失败: ' + @dbName + ' | 错误: ' + ERROR_MESSAGE()
    END CATCH

    FETCH NEXT FROM db_cursor INTO @dbName  
END  

CLOSE db_cursor  
DEALLOCATE db_cursor
```

-- 声明必要的变量
DECLARE @CurrentDataBaseName NVARCHAR(128);
DECLARE @CurrentDataBaseID INT;
DECLARE @CurrentLogName NVARCHAR(128);
DECLARE @ShrinkSize INT;
DECLARE @SQLCommand NVARCHAR(MAX);

-- 创建临时表存储数据库名称和ID
IF OBJECT_ID('tempdb..##tb_linshibiao1') IS NOT NULL
BEGIN
    DROP TABLE ##tb_linshibiao1;
END;

CREATE TABLE ##tb_linshibiao1(
    id INT IDENTITY PRIMARY KEY,
    ls_name NVARCHAR(128), 
    ls_dbid INT -- 使用 database_id 代替错误的 dbid
);

-- 插入数据库信息
INSERT INTO ##tb_linshibiao1 (ls_name, ls_dbid)
SELECT name, database_id -- 修正列名
FROM sys.databases
WHERE database_id > 4;

-- 初始化循环变量
DECLARE @TotalDBs INT = (SELECT COUNT(*) FROM ##tb_linshibiao1);
DECLARE @CurrentID INT = 1;

-- 处理每个数据库
WHILE @CurrentID <= @TotalDBs
BEGIN
    -- 获取数据库名称和ID
    SELECT @CurrentDataBaseName = ls_name, @CurrentDataBaseID = ls_dbid
    FROM ##tb_linshibiao1
    WHERE id = @CurrentID;

    -- 构建动态SQL命令
    SET @SQLCommand = N'USE [' + @CurrentDataBaseName + N'];
        ALTER DATABASE [' + @CurrentDataBaseName + N'] SET RECOVERY SIMPLE WITH NO_WAIT;
        ALTER DATABASE [' + @CurrentDataBaseName + N'] SET RECOVERY SIMPLE;
        DBCC SHRINKDATABASE (' + CAST(@CurrentDataBaseID AS NVARCHAR) + N');
        USE [master];
        ALTER DATABASE [' + @CurrentDataBaseName + N'] SET RECOVERY SIMPLE WITH NO_WAIT;';

    -- 执行动态SQL命令
    EXEC sp_executesql @SQLCommand;

    -- 打印信息，确认执行结果
    PRINT 'Executed shrink operation on database: ' + @CurrentDataBaseName;

    -- 更新循环变量
    SET @CurrentID = @CurrentID + 1;
END;

-- 清理临时表
DROP TABLE ##tb_linshibiao1;

-- 结束脚本
GO

+ <font style="color:rgb(51, 51, 51);">查询temp占用空间语句</font>

```bash
USE tempdb;
GO
SELECT name, size/128.0 AS [Size in MB], FILEPROPERTY(name, 'SpaceUsed')/128.0 AS [Space Used in MB], 
     (size - FILEPROPERTY(name, 'SpaceUsed'))/128.0 AS [Available Space in MB]
FROM sys.database_files;
```

### sql server2016 以上的脚本数据库清理以及数据库收缩
```python
SET NOCOUNT ON;
SET ANSI_WARNINGS OFF;

DECLARE @DatabaseName NVARCHAR(128),
        @LogFileName NVARCHAR(128),
        @DataFileName NVARCHAR(128),
        @SQL NVARCHAR(MAX),
        @ErrorMsg NVARCHAR(4000),
        @OriginalState NVARCHAR(60),
        @OriginalRecovery INT,
        @IsReadOnly BIT;

-- 创建错误日志表（带状态跟踪）
IF OBJECT_ID('tempdb..##LogShrink_GlobalLog') IS NOT NULL
    DROP TABLE ##LogShrink_GlobalLog;

CREATE TABLE ##LogShrink_GlobalLog (
    LogID INT IDENTITY PRIMARY KEY,
    LogTime DATETIME DEFAULT GETDATE(),
    DatabaseName NVARCHAR(128),
    DatabaseStateBefore NVARCHAR(60),
    OperationResult NVARCHAR(20),
    ErrorMessage NVARCHAR(4000)
);

-- 获取所有用户数据库（包含离线数据库）
DECLARE db_cursor CURSOR GLOBAL STATIC READ_ONLY FOR
SELECT name 
FROM sys.databases 
WHERE database_id > 4 -- 排除系统数据库
AND source_database_id IS NULL; -- 排除数据库快照

OPEN db_cursor;
FETCH NEXT FROM db_cursor INTO @DatabaseName;

WHILE @@FETCH_STATUS = 0
BEGIN
    BEGIN TRY
        -- 记录初始状态
        SELECT 
            @OriginalState = state_desc,
            @OriginalRecovery = recovery_model,
            @IsReadOnly = is_read_only
        FROM sys.databases 
        WHERE name = @DatabaseName;

        -- 跳过只读数据库
        IF @IsReadOnly = 1
        BEGIN
            INSERT INTO ##LogShrink_GlobalLog (DatabaseName, DatabaseStateBefore, OperationResult)
            VALUES (@DatabaseName, @OriginalState, 'Skipped(ReadOnly)');
            FETCH NEXT FROM db_cursor INTO @DatabaseName;
            CONTINUE;
        END

        -- 处理离线数据库
        IF @OriginalState = 'OFFLINE'
        BEGIN
            -- 尝试联机数据库（需要sysadmin权限）
            SET @SQL = N'ALTER DATABASE ' + QUOTENAME(@DatabaseName) + ' SET ONLINE;';
            EXEC sp_executesql @SQL;
            
            -- 更新状态信息
            SELECT @OriginalState = state_desc FROM sys.databases WHERE name = @DatabaseName;
        END

        -- 获取日志文件名（支持多日志文件）
        SET @LogFileName = NULL;
        SELECT TOP 1 @LogFileName = QUOTENAME(name)
        FROM sys.master_files
        WHERE database_id = DB_ID(@DatabaseName)
          AND type_desc = 'LOG'
        ORDER BY file_id;

        -- 获取数据文件名
        SET @DataFileName = NULL;
        SELECT TOP 1 @DataFileName = QUOTENAME(name)
        FROM sys.master_files
        WHERE database_id = DB_ID(@DatabaseName)
          AND type_desc = 'ROWS'
        ORDER BY file_id;

        IF @LogFileName IS NULL
        BEGIN
            RAISERROR('日志文件未找到: %s', 10, 1, @DatabaseName);
            CONTINUE;
        END

        IF @DataFileName IS NULL
        BEGIN
            RAISERROR('数据文件未找到: %s', 10, 1, @DatabaseName);
            CONTINUE;
        END

        -- 动态执行核心操作
        SET @SQL = N'
        DECLARE @InnerRecovery INT;
        BEGIN TRY
            USE master;
            
            -- 保存原始恢复模式
            SELECT @InnerRecovery = recovery_model 
            FROM sys.databases 
            WHERE name = N''' + @DatabaseName + ''';

            -- 修改恢复模式为SIMPLE
            ALTER DATABASE ' + QUOTENAME(@DatabaseName) + ' SET RECOVERY SIMPLE;

            -- 执行日志清理和收缩
            USE ' + QUOTENAME(@DatabaseName) + ';
            
            -- 清空日志
            CHECKPOINT;
            
            -- 收缩日志文件到最小
            DBCC SHRINKFILE (' + @LogFileName + ', 0) WITH NO_INFOMSGS;
            
            -- 收缩数据文件到128MB
            DBCC SHRINKDATABASE (' + QUOTENAME(@DatabaseName, '''') + ', 128) WITH NO_INFOMSGS;
            
            -- 再次收缩数据文件以确保大小
            DBCC SHRINKFILE (' + @DataFileName + ', 128) WITH NO_INFOMSGS;

            -- 恢复原始模式
            IF @InnerRecovery = 1
                ALTER DATABASE ' + QUOTENAME(@DatabaseName) + ' SET RECOVERY FULL;
            ELSE IF @InnerRecovery = 2
                ALTER DATABASE ' + QUOTENAME(@DatabaseName) + ' SET RECOVERY BULK_LOGGED;

            -- 恢复离线状态
            IF ''' + @OriginalState + ''' = ''OFFLINE''
                ALTER DATABASE ' + QUOTENAME(@DatabaseName) + ' SET OFFLINE;
        END TRY
        BEGIN CATCH
            -- 异常恢复逻辑
            IF @InnerRecovery = 1
                ALTER DATABASE ' + QUOTENAME(@DatabaseName) + ' SET RECOVERY FULL;
            ELSE IF @InnerRecovery = 2
                ALTER DATABASE ' + QUOTENAME(@DatabaseName) + ' SET RECOVERY BULK_LOGGED;

            IF ''' + @OriginalState + ''' = ''OFFLINE''
                ALTER DATABASE ' + QUOTENAME(@DatabaseName) + ' SET OFFLINE;

            THROW;
        END CATCH';

        EXEC sp_executesql @SQL;
        
        INSERT INTO ##LogShrink_GlobalLog (DatabaseName, DatabaseStateBefore, OperationResult)
        VALUES (@DatabaseName, @OriginalState, 'Success');

    END TRY
    BEGIN CATCH
        SET @ErrorMsg = ERROR_MESSAGE();
        INSERT INTO ##LogShrink_GlobalLog (DatabaseName, DatabaseStateBefore, OperationResult, ErrorMessage)
        VALUES (@DatabaseName, @OriginalState, 'Failed', @ErrorMsg);
    END CATCH

    FETCH NEXT FROM db_cursor INTO @DatabaseName;
END

CLOSE db_cursor;
DEALLOCATE db_cursor;

-- 显示操作结果
SELECT 
    LogTime,
    DatabaseName,
    DatabaseStateBefore,
    OperationResult,
    ErrorMessage
FROM ##LogShrink_GlobalLog
ORDER BY LogID;
```

### 定义数据库清理
```python
SET NOCOUNT ON;
GO

-- 声明变量
DECLARE @DatabaseName NVARCHAR(128),
        @SQL NVARCHAR(MAX),
        @ExcludeDB NVARCHAR(128) = 'dyjy2';  -- 需要保留的数据库

-- 创建日志表
IF OBJECT_ID('tempdb..#DeleteLog') IS NOT NULL
    DROP TABLE #DeleteLog;

CREATE TABLE #DeleteLog (
    DBName NVARCHAR(128),
    Status NVARCHAR(50),
    ErrorMessage NVARCHAR(4000),
    ExecuteTime DATETIME DEFAULT GETDATE()
);

-- 生成删除列表
DECLARE db_cursor CURSOR LOCAL FAST_FORWARD FOR
SELECT name 
FROM sys.databases 
WHERE name NOT IN (
    'master', 'model', 'msdb', 'tempdb',  -- 系统数据库
    @ExcludeDB                            -- 需要保留的数据库
)
AND state = 0  -- 只处理在线数据库
AND source_database_id IS NULL;  -- 排除数据库快照

OPEN db_cursor;
FETCH NEXT FROM db_cursor INTO @DatabaseName;

WHILE @@FETCH_STATUS = 0
BEGIN
    BEGIN TRY
        -- 生成清理连接命令
        SET @SQL = N'
        ALTER DATABASE ' + QUOTENAME(@DatabaseName) + ' SET SINGLE_USER WITH ROLLBACK IMMEDIATE;
        DROP DATABASE ' + QUOTENAME(@DatabaseName) + ';
        ';

        EXEC sp_executesql @SQL;
        
        INSERT INTO #DeleteLog (DBName, Status)
        VALUES (@DatabaseName, 'Success');
    END TRY
    BEGIN CATCH
        INSERT INTO #DeleteLog (DBName, Status, ErrorMessage)
        VALUES (@DatabaseName, 'Failed', ERROR_MESSAGE());
    END CATCH

    FETCH NEXT FROM db_cursor INTO @DatabaseName;
END;

CLOSE db_cursor;
DEALLOCATE db_cursor;

-- 显示操作结果
SELECT * FROM #DeleteLog;
GO
```
