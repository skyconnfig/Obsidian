# 易考通查询试卷下面的题型得分

### 查询考试 id
```plain
declare @p7 int
set @p7=58
exec proc_Common_PageList  @tab='(select ERId,ER_EId,ER_PId,ER_MId,E_Name,(case P_Kind when 1 then ''系统试卷''  when 2 then ''讲师试卷'' end) as  P_Kind,
(select UserName from tb_UserInfo WITH(NOLOCK) where UId=E_AddOperator) as AUserName,UR_Custom2,(case E_Type when 1 then ''考试模式''  when 2 then ''练习模式'' end) as E_Type,
UserNo,UserName,Name,TeamName,e.Custom2,
Round((select SUM(EP_Score) from tb_HB_ExaminationPapers WITH(NOLOCK) where EP_PId=PId),2) as Score,ER_Score,ER_AddTime,ISNULL((select top 1 Max(ER_Score) from tb_ExaminationResult WITH(NOLOCK) where ER_EId=a.ER_EId and ER_PId=a.ER_PId and ER_MId=a.ER_MId and ER_Type=1),0) as TotalScore from tb_ExaminationResult a WITH(NOLOCK) 
inner join tb_HB_Examination b WITH(NOLOCK) on a.ER_EId=b.EId
inner join tb_HB_Paper c WITH(NOLOCK) on c.PId=b.E_PId
inner join tb_UserInfo d WITH(NOLOCK) on d.UId=a.ER_MId
inner join tb_Team e WITH(NOLOCK) on e.Id=d.UserClassId
inner join tb_School f WITH(NOLOCK) on f.Id=e.SchoolId where 1=1 and ER_Type=1  and ER_State=0 and E_Type in(1,2)  and  b.E_AddOperator = 3 and E_Name like ''%邱县反假币知识竞赛%'') t',@strFld=' *',@strWhere=' 1=1 ',@PageIndex =1,@PageSize=60,@Sort='ER_Score desc',@RecordCount=@p7 output,@IsGetCount=0
select @p7

```

```plain


SELECT 
    d.ED_MId,
    d.ED_Custom2 AS 账号,  -- 添加账号字段
    q.QB_Type AS 题目类型,
    SUM(e.EP_Score) AS 单项总分,
    COUNT(q.QuestionBId) AS 题目数量,
    ISNULL(SUM(d.ED_Goal), 0) AS 得分合计
FROM tb_HB_ExaminationPapers e WITH(NOLOCK)  
INNER JOIN tb_HB_QuestionBank q WITH(NOLOCK) ON e.EP_QBId = q.QuestionBId
INNER JOIN tb_ExaminationDetails d WITH(NOLOCK) ON d.ED_QBId = q.QuestionBId
    AND d.ED_PId = '16113' 
    AND d.ED_EId = 5279 
    AND d.ED_State = 1 
WHERE e.EP_PId = '16113' 
GROUP BY d.ED_MId, d.ED_Custom2, q.QB_Type  -- 添加ED_Custom2到GROUP BY
ORDER BY d.ED_MId, q.QB_Type
```
