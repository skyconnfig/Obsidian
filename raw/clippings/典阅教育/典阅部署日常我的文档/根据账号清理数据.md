DECLARE test_Cursor CURSOR LOCAL FOR    
SELECT name

    -- 执行DELETE语句  
    EXEC sp_executesql @sql     
      
    FETCH NEXT FROM test_Cursor INTO @tablename    
END    

CLOSE test_Cursor    
DEALLOCATE test_Cursor
