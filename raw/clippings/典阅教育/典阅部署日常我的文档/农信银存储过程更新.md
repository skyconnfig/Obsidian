# 农信银存储过程更新

```powershell
/*
==============================================================================
对象名称: [dbo].[GetRace]
描述:     获取学生/员工竞赛列表通用存储过程
功能:     1. 自动激活长期未更新的任务(重置开始时间为今年1月1日)
          2. 根据 @Otype 返回不同类型的竞赛列表
兼容类型: 0-9 (涵盖点钞,传票,字符,零售,对公,机器点钞,Excel,全能等)
==============================================================================
*/

SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO

CREATE OR ALTER PROCEDURE [dbo].[GetRace] 
(
   @BankSiteId int,      -- 站点ID / 机构ID
   @Otype int            -- 业务类型:
                         -- 0: 机算传票(非全能/非珠算)
                         -- 1: 点钞(非全能)
                         -- 2: 珠算传票(非全能)
                         -- 3: 字符(非全能)
                         -- 4: 零售业务
                         -- 5: 对公业务
                         -- 6: 综合考试(非全能, 含试卷入口)
                         -- 7: 全能综合考试(含全能项, 机器点钞, Excel等)
                         -- 8: 机器点钞(非全能)
                         -- 9: Excel制表(非全能)
)
AS
BEGIN
    SET NOCOUNT ON;

    -- ==========================================
    -- 第一部分：数据预处理 (自动激活长期任务)
    -- 目的：确保时长异常大(>1000000分钟)且过期的任务，能重新进入"最近11分钟"的筛选窗口
    -- ==========================================
    DECLARE @CurrentYearStart DATE;
    SET @CurrentYearStart = CAST(CAST(YEAR(GETDATE()) AS VARCHAR(4)) + '-01-01' AS DATE);

    BEGIN TRY
        -- 1. 复杂业务计时器 (零售/对公) - 字段: BeginTime, ExamLength
        IF EXISTS (SELECT 1 FROM sys.objects WHERE name = 'dal_ComplexTimer' AND type = 'U')
        BEGIN
            UPDATE dbo.dal_ComplexTimer 
            SET BeginTime = @CurrentYearStart
            WHERE BeginTime < @CurrentYearStart 
              AND ExamLength > 1000000;
        END

        -- 2. 字符录入 - 字段: BeginTime, CharacterLength
        IF EXISTS (SELECT 1 FROM sys.objects WHERE name = 'dal_Character' AND type = 'U')
        BEGIN
            UPDATE dbo.dal_Character 
            SET BeginTime = @CurrentYearStart
            WHERE BeginTime < @CurrentYearStart 
              AND CharacterLength > 1000000;
        END

        -- 3. 传票算 - 字段: BeginTime, SummonCountLength
        IF EXISTS (SELECT 1 FROM sys.objects WHERE name = 'dal_SummonCount' AND type = 'U')
        BEGIN
            UPDATE dbo.dal_SummonCount 
            SET BeginTime = @CurrentYearStart
            WHERE BeginTime < @CurrentYearStart 
              AND SummonCountLength > 1000000;
        END

        -- 4. 点钞 - 字段: BeginTime, CountingLength
        IF EXISTS (SELECT 1 FROM sys.objects WHERE name = 'dal_Counting' AND type = 'U')
        BEGIN
            UPDATE dbo.dal_Counting 
            SET BeginTime = @CurrentYearStart
            WHERE BeginTime < @CurrentYearStart 
              AND CountingLength > 1000000;
        END

        -- 5. 机器点钞 - 字段: StartTime, AppraisalTime
        IF EXISTS (SELECT 1 FROM sys.objects WHERE name = 'dal_MachineBill' AND type = 'U')
        BEGIN
            UPDATE dbo.dal_MachineBill 
            SET StartTime = @CurrentYearStart
            WHERE StartTime < @CurrentYearStart 
              AND AppraisalTime > 1000000;
        END

        -- 6. Excel制表 - 字段: StartTime, ExaminationTime
        IF EXISTS (SELECT 1 FROM sys.objects WHERE name = 'Excel_Tabulation' AND type = 'U')
        BEGIN
            UPDATE dbo.Excel_Tabulation 
            SET StartTime = @CurrentYearStart
            WHERE StartTime < @CurrentYearStart 
              AND ExaminationTime > 1000000;
        END
    END TRY
    BEGIN CATCH
        -- 数据修复失败不影响查询，仅记录或忽略，确保主业务流程不中断
        -- PRINT ERROR_MESSAGE(); 
    END CATCH

    -- ==========================================
    -- 第二部分：业务查询逻辑
    -- ==========================================

    -- 0. 机算传票 (非全能, 非珠算)
    IF (@Otype = 0)
    BEGIN
        SELECT id, SummonCountName AS [text], 2 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/cp.png' AS icon,
               'CP/ExamCP.aspx?examId=' + CAST(id AS VARCHAR) AS href 
        FROM dbo.dal_SummonCount 	
        WHERE BankSiteId = @BankSiteId 
          AND SummonCountStatus = 1 
          AND DATEDIFF(mi, GETDATE(), BeginTime) < 11	
          AND DATEADD(mi, SummonCountLength, BeginTime) > GETDATE() 
          AND SummonCountName NOT LIKE '%全能%' 
          AND SummonCountName NOT LIKE '%珠算%';
    END 

    -- 1. 点钞比赛 (非全能)
    IF (@Otype = 1)
    BEGIN
        SELECT id, CountingName AS [text], 1 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/dc.png' AS icon,
               'DC/ExamDC4.aspx?examId=' + CAST(id AS VARCHAR) AS href 
        FROM dbo.dal_Counting 
        WHERE BankSiteId = @BankSiteId  
          AND CountingStatus = 1 
          AND DATEDIFF(mi, GETDATE(), BeginTime) < 11 
          AND DATEADD(mi, CountingLength, BeginTime) > GETDATE()
          AND CountingName NOT LIKE '%全能%';
    END

    -- 2. 珠算传票 (非全能, 含珠算)
    IF (@Otype = 2)
    BEGIN
        SELECT id, SummonCountName AS [text], 2 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/cp.png' AS icon,
               'CP/ExamCP.aspx?examId=' + CAST(id AS VARCHAR) AS href 
        FROM dbo.dal_SummonCount 	
        WHERE BankSiteId = @BankSiteId 
          AND SummonCountStatus = 1 
          AND DATEDIFF(mi, GETDATE(), BeginTime) < 11	
          AND DATEADD(mi, SummonCountLength, BeginTime) > GETDATE()
          AND SummonCountName NOT LIKE '%全能%' 
          AND SummonCountName LIKE '%珠算%';
    END 

    -- 3. 字符录入 (非全能)
    IF (@Otype = 3)
    BEGIN
        SELECT id, CharacterName AS [text], 3 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/zf.png' AS icon,
               'DZ/ExamDZ.aspx?examId=' + CAST(id AS VARCHAR) AS href 
        FROM dbo.dal_Character
        WHERE BankSiteId = @BankSiteId 
          AND CharacterStatus = 1 
          AND DATEDIFF(mi, GETDATE(), BeginTime) < 11 
          AND DATEADD(mi, CharacterLength, BeginTime) > GETDATE()
          AND CharacterName NOT LIKE '%全能%';
    END

    -- 4. 零售业务
    IF (@Otype = 4)
    BEGIN
        SELECT a.id, b.Plan_Name AS [text], 4 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/ls.png' AS icon,
               '/Bank/Admin?Optype=1&examId=' + CAST(a.id AS VARCHAR) + '&planId=' + CAST(b.id AS VARCHAR) + '&nxyplanid=' + CAST(ISNULL(b.nxyPlanID,0) AS VARCHAR) AS href 
        FROM dbo.dal_ComplexTimer a 
        INNER JOIN dbo.dal_ComplexPlan b ON a.ComplexPlanId = b.Id
        WHERE a.BankSiteId = @BankSiteId 
          AND a.ExamStatus = 1 	
          AND DATEDIFF(mi, GETDATE(), a.BeginTime) < 11 
          AND DATEADD(mi, a.ExamLength, a.BeginTime) > GETDATE()	
          AND b.Plan_Name LIKE '%零售%';
    END

    -- 5. 对公业务
    IF (@Otype = 5)
    BEGIN
        SELECT a.id, b.Plan_Name AS [text], 4 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/dg.png' AS icon,
               '/Bank/Admin?Optype=2&examId=' + CAST(a.id AS VARCHAR) + '&planId=' + CAST(b.id AS VARCHAR) + '&nxyplanid=' + CAST(ISNULL(b.nxyPlanID,0) AS VARCHAR) AS href 
        FROM dbo.dal_ComplexTimer a 
        INNER JOIN dbo.dal_ComplexPlan b ON a.ComplexPlanId = b.Id
        WHERE a.BankSiteId = @BankSiteId 
          AND a.ExamStatus = 1 	
          AND DATEDIFF(mi, GETDATE(), a.BeginTime) < 11 
          AND DATEADD(mi, a.ExamLength, a.BeginTime) > GETDATE()	
          AND b.Plan_Name LIKE '%对公%';
    END

    -- 6. 综合集合 (非全能, 含试卷入口)
    IF (@Otype = 6)
    BEGIN
        -- 固定入口：试卷测试
        SELECT 1 AS id, '试卷测试' AS [text], 4 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/cs.png' AS icon,
               'Bank/Admin?Optype=9&examId=1&planId=1' AS href
        UNION ALL 
        -- 点钞
        SELECT id, CountingName AS [text], 1 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/dc.png' AS icon,
               'DC/ExamDC.aspx?examId=' + CAST(id AS VARCHAR) AS href 
        FROM dbo.dal_Counting 
        WHERE BankSiteId = @BankSiteId 	
          AND CountingStatus = 1 
          AND DATEDIFF(mi, GETDATE(), BeginTime) < 11 
          AND DATEADD(mi, CountingLength, BeginTime) > GETDATE()
        UNION ALL
        -- 传票
        SELECT id, SummonCountName AS [text], 2 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/cp.png' AS icon,
               'CP/ExamCP.aspx?examId=' + CAST(id AS VARCHAR) AS href 
        FROM dbo.dal_SummonCount 	
        WHERE BankSiteId = @BankSiteId 
          AND SummonCountStatus = 1 
          AND DATEDIFF(mi, GETDATE(), BeginTime) < 11	
          AND DATEADD(mi, SummonCountLength, BeginTime) > GETDATE()
        UNION ALL
        -- 字符
        SELECT id, CharacterName AS [text], 3 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/zf.png' AS icon,
               'DZ/ExamDZ.aspx?examId=' + CAST(id AS VARCHAR) AS href 
        FROM dbo.dal_Character
        WHERE BankSiteId = @BankSiteId 
          AND CharacterStatus = 1 
          AND DATEDIFF(mi, GETDATE(), BeginTime) < 11 
          AND DATEADD(mi, CharacterLength, BeginTime) > GETDATE()
        UNION ALL 
        -- 零售
        SELECT a.id, b.Plan_Name AS [text], 4 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/ls.png' AS icon,
               '/Bank/Admin?Optype=1&examId=' + CAST(a.id AS VARCHAR) + '&planId=' + CAST(b.id AS VARCHAR) AS href 
        FROM dbo.dal_ComplexTimer a 
        INNER JOIN dbo.dal_ComplexPlan b ON a.ComplexPlanId = b.Id
        WHERE a.BankSiteId = @BankSiteId 
          AND a.ExamStatus = 1 	
          AND DATEDIFF(mi, GETDATE(), a.BeginTime) < 11 
          AND DATEADD(mi, a.ExamLength, a.BeginTime) > GETDATE()	
          AND b.Plan_Name LIKE '%零售%' 
          AND b.Plan_Name NOT LIKE '%练习%'
        UNION ALL
        -- 对公
        SELECT a.id, b.Plan_Name AS [text], 4 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/dg.png' AS icon,
               '/Bank/Admin?Optype=2&examId=' + CAST(a.id AS VARCHAR) + '&planId=' + CAST(b.id AS VARCHAR) AS href 
        FROM dbo.dal_ComplexTimer a 
        INNER JOIN dbo.dal_ComplexPlan b ON a.ComplexPlanId = b.Id
        WHERE a.BankSiteId = @BankSiteId 
          AND a.ExamStatus = 1 	
          AND DATEDIFF(mi, GETDATE(), a.BeginTime) < 11 
          AND DATEADD(mi, a.ExamLength, a.BeginTime) > GETDATE()	
          AND b.Plan_Name LIKE '%对公%' 
          AND b.Plan_Name NOT LIKE '%练习%';
    END

    -- 7. 全能综合集合 (包含全能项, 机器点钞, Excel等)
    IF (@Otype = 7)
    BEGIN
        -- 全能点钞
        SELECT id, CountingName AS [text], 1 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/dc.png' AS icon,
               'DC/ExamDC4.aspx?examId=' + CAST(id AS VARCHAR) AS href 
        FROM dbo.dal_Counting 
        WHERE BankSiteId = @BankSiteId 	
          AND CountingStatus = 1 
          AND DATEDIFF(mi, GETDATE(), BeginTime) < 11 
          AND DATEADD(mi, CountingLength, BeginTime) > GETDATE() 
          AND CountingName LIKE '%全能%' 
        
        UNION ALL
        
        -- 全能机器点钞
        SELECT id, BanknoteName AS [text], 1 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/dc.png' AS icon,
               'DC/ExamJQDC.aspx?examId=' + CAST(id AS VARCHAR) AS href 
        FROM dbo.dal_MachineBill 
        WHERE GeneralId = @BankSiteId 	
          AND State = 0 
          AND DATEDIFF(mi, GETDATE(), StartTime) < 11 
          AND DATEADD(mi, AppraisalTime, StartTime) > GETDATE() 
          AND BanknoteName LIKE '%全能%' 
        
        UNION ALL
        
        -- 全能机算传票 (非珠算)
        SELECT id, SummonCountName AS [text], 2 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/cp.png' AS icon,
               'CP/ExamCP.aspx?examId=' + CAST(id AS VARCHAR) AS href 
        FROM dbo.dal_SummonCount 	
        WHERE BankSiteId = @BankSiteId 
          AND SummonCountStatus = 1 
          AND DATEDIFF(mi, GETDATE(), BeginTime) < 11	
          AND DATEADD(mi, SummonCountLength, BeginTime) > GETDATE()  
          AND SummonCountName LIKE '%全能%'  
          AND SummonCountName NOT LIKE '%珠算%' 
        
        UNION ALL
        
        -- 全能珠算传票
        SELECT id, SummonCountName AS [text], 2 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/cp.png' AS icon,
               'CP/ExamCP.aspx?examId=' + CAST(id AS VARCHAR) AS href 
        FROM dbo.dal_SummonCount 	
        WHERE BankSiteId = @BankSiteId 
          AND SummonCountStatus = 1 
          AND DATEDIFF(mi, GETDATE(), BeginTime) < 11	
          AND DATEADD(mi, SummonCountLength, BeginTime) > GETDATE()  
          AND SummonCountName LIKE '%全能%'  
          AND SummonCountName LIKE '%珠算%' 
        
        UNION ALL
        
        -- 全能字符
        SELECT id, CharacterName AS [text], 3 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/zf.png' AS icon,
               'DZ/ExamDZ.aspx?examId=' + CAST(id AS VARCHAR) AS href 
        FROM dbo.dal_Character
        WHERE BankSiteId = @BankSiteId 
          AND CharacterStatus = 1 
          AND DATEDIFF(mi, GETDATE(), BeginTime) < 11
          AND DATEADD(mi, CharacterLength, BeginTime) > GETDATE()  
          AND CharacterName LIKE '%全能%' 
        
        UNION ALL 
        
        -- 全能零售
        SELECT a.id, b.Plan_Name AS [text], 4 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/ls.png' AS icon,
               '/Bank/Admin?Optype=1&examId=' + CAST(a.id AS VARCHAR) + '&planId=' + CAST(b.id AS VARCHAR) AS href 
        FROM dbo.dal_ComplexTimer a 
        INNER JOIN dbo.dal_ComplexPlan b ON a.ComplexPlanId = b.Id
        WHERE a.BankSiteId = @BankSiteId 
          AND a.ExamStatus = 1 	
          AND DATEDIFF(mi, GETDATE(), a.BeginTime) < 11 
          AND DATEADD(mi, a.ExamLength, a.BeginTime) > GETDATE()	
          AND b.Plan_Name LIKE '%零售%' 
          AND b.Plan_Name NOT LIKE '%练习%'   
          AND b.Plan_Name LIKE '%全能%' 
        
        UNION ALL
        
        -- 全能对公
        SELECT a.id, b.Plan_Name AS [text], 4 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/dg.png' AS icon,
               '/Bank/Admin?Optype=2&examId=' + CAST(a.id AS VARCHAR) + '&planId=' + CAST(b.id AS VARCHAR) AS href 
        FROM dbo.dal_ComplexTimer a 
        INNER JOIN dbo.dal_ComplexPlan b ON a.ComplexPlanId = b.Id
        WHERE a.BankSiteId = @BankSiteId 
          AND a.ExamStatus = 1 	
          AND DATEDIFF(mi, GETDATE(), a.BeginTime) < 11 
          AND DATEADD(mi, a.ExamLength, a.BeginTime) > GETDATE()	
          AND b.Plan_Name LIKE '%对公%' 
          AND b.Plan_Name NOT LIKE '%练习%'  
          AND b.Plan_Name LIKE '%全能%'
          
        UNION ALL
        
        -- 全能Excel制表
        SELECT id, AppraisalName AS [text], 1 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/dc.png' AS icon,
               'ZB/ExcelZB.aspx?examId=' + CAST(id AS VARCHAR) AS href 
        FROM dbo.Excel_Tabulation 
        WHERE GeneralLineID = @BankSiteId 	
          AND CountingStatus = 1 
          AND DATEDIFF(mi, GETDATE(), StartTime) < 11 
          AND DATEADD(mi, ExaminationTime, StartTime) > GETDATE() 
          AND AppraisalName LIKE '%全能%';
    END

    -- 8. 机器点钞 (非全能)
    IF (@Otype = 8)
    BEGIN
        SELECT id, BanknoteName AS [text], 1 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/dc.png' AS icon,
               'DC/ExamJQDC.aspx?examId=' + CAST(id AS VARCHAR) AS href 
        FROM dbo.dal_MachineBill 
        WHERE GeneralId = @BankSiteId 	
          AND State = 0 
          AND DATEDIFF(mi, GETDATE(), StartTime) < 11 
          AND BanknoteName NOT LIKE '%全能%';
    END

    -- 9. Excel制表 (非全能)
    IF (@Otype = 9)
    BEGIN
        SELECT id, AppraisalName AS [text], 1 AS [type], 
               '/Scripts/Desktop/themes/windows/icons/dc.png' AS icon,
               'ZB/ExcelZB.aspx?examId=' + CAST(id AS VARCHAR) AS href 
        FROM dbo.Excel_Tabulation 
        WHERE GeneralLineID = @BankSiteId 	
          AND CountingStatus = 1 
          AND DATEDIFF(mi, GETDATE(), StartTime) < 11 
          AND DATEADD(mi, ExaminationTime, StartTime) > GETDATE() 
          AND AppraisalName NOT LIKE '%全能%';
    END 

END
GO
```
