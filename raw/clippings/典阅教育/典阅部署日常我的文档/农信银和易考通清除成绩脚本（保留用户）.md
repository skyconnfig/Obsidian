DECLARE test_Cursor CURSOR LOCAL FOR
select name from dbo.sysobjects where xtype='u' and (not name LIKE 'dtproperties') and name like'yw_%' order by name
open test_Cursor

FETCH NEXT FROM test_Cursor into @tablename
WHILE  (@@fetch_status=0)

begin
exec('truncate table '+ @tablename)
fetch next from test_Cursor into @tablename
end
close test_Cursor

DELETE FROM tb_ExaminationResult 

WHERE ER_EId = 0 

  AND ER_State = 0 

  AND ER_Type = 1

  AND ER_MId IN (

    SELECT UId FROM tb_UserInfo a 

    INNER JOIN tb_Team b ON b.Id = a.UserClassId

    INNER JOIN tb_School c ON c.Id = b.SchoolId

    WHERE UserType = 3 AND State = 1 AND a.AddOperator = 3

  );



-- 2. 删除CFA结果数据

DELETE m FROM tb_HB_CFAResult m 

INNER JOIN tb_HB_CFARound n ON m.R_RounId = n.Id AND m.R_CFAId = n.CFAId 

WHERE R_CFAId = 0 

  AND R_C_MId IN (

    SELECT UId FROM tb_UserInfo a 

    INNER JOIN tb_Team b ON b.Id = a.UserClassId

    INNER JOIN tb_School c ON c.Id = b.SchoolId

    WHERE UserType = 3 AND State = 1 AND a.AddOperator = 3

  );
