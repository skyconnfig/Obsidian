### 比赛全部数据库备份脚本
```sql
DECLARE @backupPath NVARCHAR(255) = 'C:\dbBank\江苏省赛\制作账号' -- 修改为你的备份路径
DECLARE @timestamp VARCHAR(20) = CONVERT(VARCHAR(20), GETDATE(), 112) + REPLACE(CONVERT(VARCHAR(20), GETDATE(), 108), ':', '')
DECLARE @dbName NVARCHAR(255)
DECLARE @sqlCommand NVARCHAR(MAX)

DECLARE db_cursor CURSOR FOR
SELECT name 
FROM sys.databases 
WHERE name NOT IN ('master', 'model', 'msdb', 'tempdb')  -- 排除系统数据库
AND state = 0  -- 只处理在线数据库

OPEN db_cursor
FETCH NEXT FROM db_cursor INTO @dbName

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @sqlCommand = '
    BACKUP DATABASE [' + @dbName + '] 
    TO DISK = ''' + @backupPath + @dbName + '_' + @timestamp + '.bak''
    WITH COMPRESSION,  -- 启用压缩（需要企业版/标准版）
    CHECKSUM,          -- 验证校验和
    STATS = 5;         -- 显示进度（每5%更新）
    '

    PRINT 'Backing up: ' + @dbName
    EXEC(@sqlCommand)
    
    FETCH NEXT FROM db_cursor INTO @dbName
END

CLOSE db_cursor
```
