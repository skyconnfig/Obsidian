```plain
-- 声明变量
DECLARE @StartNum INT = 1;
DECLARE @EndNum INT = 400;

-- 请根据实际情况修改这些值
DECLARE @AddUserId INT = 1; -- 执行插入操作的用户ID
DECLARE @SchoolId INT = 100;   -- 班级所属的学校ID
DECLARE @CollegeId INT = 200;  -- 班级所属的学院ID
DECLARE @MajorId INT = 300;    -- 班级所属的专业ID
DECLARE @TeacherId INT = 999;  -- 指导老师的UserId
DECLARE @Grade VARCHAR(50) = NULL; -- 年级，例如 '2023级'，或保持 NULL

-- 使用递归CTE生成从 3 到 400 的序列
WITH NumberSeries AS (
    SELECT @StartNum AS num
    UNION ALL
    SELECT num + 1
    FROM NumberSeries
    WHERE num < @EndNum
)

-- 将生成的序列插入到 tb_Class 表
INSERT INTO tb_Class (
    SchoolId,
    CollegeId,
    MajorId,
    Grade,         -- 根据表结构添加
    ClassName,
    TeacherId,
    AddUserId,
    AddTime,
    -- Type 列不存在，已移除
    start_time,
    end_time
)
SELECT 
    @SchoolId,
    @CollegeId,
    @MajorId,
    @Grade,                             -- 对应的值
    '代表队' + CAST(num AS VARCHAR(10)), -- 生成 "代表队1", "代表队2", ... "代表队400"
    @TeacherId,
    @AddUserId,
    GETDATE(),                          -- 使用当前时间作为 AddTime
    NULL,                               -- start_time
    NULL                                -- end_time
FROM NumberSeries
OPTION (MAXRECURSION 0); -- 允许递归到400

-- 查看成功插入的记录数
SELECT @@ROWCOUNT AS '成功插入的记录数';
```

```plain
-- -----------------------------
-- 步骤 0: 声明全局变量
-- -----------------------------
DECLARE @StartClassId INT = 1;
DECLARE @EndClassId INT = 400;
DECLARE @StudentsPerClass INT = 60;
DECLARE @TotalStudents INT = (@EndClassId - @StartClassId + 1) * @StudentsPerClass; -- 24000

-- 通用配置
DECLARE @AddUserId INT = 1;          -- 执行此操作的管理员ID
DECLARE @DefaultPassword VARCHAR(50) = '888888'; -- 统一密码

-- -----------------------------
-- 步骤 1: 创建一个包含 1 到 24000 序号的 CTE
-- -----------------------------
WITH NumberSeries AS (
    SELECT 1 AS num
    UNION ALL
    SELECT num + 1
    FROM NumberSeries
    WHERE num < @TotalStudents
)

-- -----------------------------
-- 步骤 2: 批量插入用户 (tb_User)
-- 登录账号格式: s00001, s00002, ..., s24000
-- -----------------------------
INSERT INTO tb_User (LoginNo, Password, State, AddTime)
SELECT 
    's' + RIGHT('00000' + CAST(num AS VARCHAR(5)), 5), -- 生成 s00001 格式
    @DefaultPassword,
    1, -- 状态: 1=正常
    GETDATE()
FROM NumberSeries
OPTION (MAXRECURSION 0);

-- -----------------------------
-- 步骤 3: 批量插入学生 (tb_Student)
-- 将 24000 个用户平均分配到 400 个班级 (C_ID)
-- -----------------------------
WITH NumberSeries AS (
    SELECT 1 AS num
    UNION ALL
    SELECT num + 1
    FROM NumberSeries
    WHERE num < @TotalStudents
)
INSERT INTO tb_Student (
    SchoolId,
    CollegeId,
    MajorId,
    ClassId,  -- 即 C_ID
    UserId,   -- 关联 tb_User.U_ID
    StudentNo,
    Name,
    AddUserId,
    AddTime
)
SELECT 
    -- 从 tb_Class 表获取该班级的 SchoolId, CollegeId, MajorId
    c.SchoolId,
    c.CollegeId,
    c.MajorId,
    c.C_ID AS ClassId,
    -- 关联用户: 学号 s00001 对应 U_ID=1, 以此类推
    u.U_ID AS UserId,
    u.LoginNo AS StudentNo,
    '学生' + CAST(ns.num AS VARCHAR(5)) AS Name,
    @AddUserId,
    GETDATE()
FROM NumberSeries ns
-- 计算每个学生所属的班级 (C_ID)
CROSS APPLY (
    SELECT 
        @StartClassId + (ns.num - 1) / @StudentsPerClass AS ClassId_for_num
) calc
-- 与 tb_Class 表连接，获取班级的 SchoolId, CollegeId, MajorId
JOIN tb_Class c ON c.C_ID = calc.ClassId_for_num
-- 与 tb_User 表连接，获取对应的用户ID
JOIN tb_User u ON u.LoginNo = 's' + RIGHT('00000' + CAST(ns.num AS VARCHAR(5)), 5)
OPTION (MAXRECURSION 0);

-- -----------------------------
-- 步骤 4: 验证结果
-- -----------------------------
SELECT 
    COUNT(*) AS Total_Users_Inserted,
    (SELECT COUNT(*) FROM tb_Student) AS Total_Students_Inserted
FROM tb_User;
```
