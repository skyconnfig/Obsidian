# 比赛导出成绩脚本

### 1.字符导出成绩脚本
```bash
select t3.CharacterName as '考核名称',t1.Teller_Name as '选手姓名',t1.Teller_No as '选手账号',t4.TeamName as '代表队',t2.Result as '得分',
t2.AddTime as '提交时间',DATEDIFF(s,t3.BeginTime,t2.AddTime) as '用时(秒)'
from bi_BankTeller as t1
inner join dzs_ExamResult as t2 on t1.id=t2.TellerId
inner join dal_Character as t3 on t2.ExamId=t3.Id
inner join bi_Team as t4 on t1.Team_Id=t4.Id
order by t2.Result desc,DATEDIFF(s,t2.AddTime,t2.AddTime) asc
```

### 3.对公零售导出成绩脚本

### 4.机器点钞导出成绩脚本
```plsql

--机器点钞
select t3.[BanknoteName] as '考核名称',t1.Teller_Name as '选手姓名',t1.Teller_No as '选手账号',t4.TeamName as '代表队',
(case when t2.[FinalScore]-isnull(t2.KouFen,0)<0 then 0 else t2.[FinalScore]-isnull(t2.KouFen,0) end)  as '得分',
t2.AddTime as '提交时间',(case when DATEDIFF(s,t3.[StartTime],t2.AddTime)>600 then 600 else DATEDIFF(s,t3.[StartTime],t2.AddTime) end) as '用时(秒)'
from bi_BankTeller as t1
inner join dms_ExamResult as t2 on t1.id=t2.TellerId
inner join [dbo].[dal_MachineBill] as t3 on t2.ExamId=t3.Id
inner join bi_Team as t4 on t1.Team_Id=t4.Id
where t2.ExamId=267
order by (case when t2.[FinalScore]-isnull(t2.KouFen,0)<0 then 0 else t2.[FinalScore]-isnull(t2.KouFen,0) end) desc,
(case when DATEDIFF(s,t3.[StartTime],t2.AddTime)>600 then 600 else DATEDIFF(s,t3.[StartTime],t2.AddTime) end) asc
```

### 5.理论知识成绩导出脚本
```plsql
select t3.E_Name as '考核名称',t1.UserName as  '选手姓名',t1.UserNo as '选手账号',t4.TeamName as '代表队名称',
t2.ER_Score as '成绩',t2.ER_AddTime as '提交时间', t2.ER_Custom3 as '用时(分)',cast(substring(t2.ER_Custom3,0,charindex(':',left(t2.ER_Custom3,3))) as int)*3600
+cast(substring(right(t2.ER_Custom3,5),0,charindex(':',right(t2.ER_Custom3,5))) as int)*60
+substring(right(t2.ER_Custom3,5),charindex(':',right(t2.ER_Custom3,5))+1,len(right(t2.ER_Custom3,5))-charindex(':',right(t2.ER_Custom3,5)))  as '用时(秒)'
from tb_UserInfo as t1
left join tb_ExaminationResult as t2 on t1.UserNo=t2.ER_Custom2
inner join tb_HB_Examination as t3 on t2.ER_EId=t3.EId and t2.ER_PId=t3.E_PId
inner join tb_Team as t4 on t1.UserClassId=t4.Id
order by t2.ER_Score desc,cast(substring(t2.ER_Custom3,0,charindex(':',left(t2.ER_Custom3,3))) as int)*3600
+cast(substring(right(t2.ER_Custom3,5),0,charindex(':',right(t2.ER_Custom3,5))) as int)*60
+substring(right(t2.ER_Custom3,5),charindex(':',right(t2.ER_Custom3,5))+1,len(right(t2.ER_Custom3,5))-charindex(':',right(t2.ER_Custom3,5))) asc
```

### 7.厅堂服务成绩导出脚本
```plsql
select t5.PracticeName as '考核名称', t3.Name as '选手姓名',t1.LoginNo as '选手账号',t4.ClassName as '代表队名称',t2.Scores as '成绩',t2.UpdateTime as '提交时间',
(case when t2.Tstate=1 then '已提交' else '未提交' end) as '状态',DATEDIFF(s,t2.AddTime,t2.UpdateTime) as '用时(秒)' from tb_User as t1
left join bsi_TotalResult as t2 on t1.U_ID=t2.UserId
inner join tb_Student as t3 on t3.UserId=t1.U_ID
inner join tb_Class as t4 on t4.C_ID=t3.ClassId
inner join bsi_PracticeAssessment as t5 on t2.ExamId=t5.ID
--where t2.ExamId=3 
--and (case when t2.Tstate=1 then '已提交' else '未提交' end)='未提交'
order by t2.Scores desc ,DATEDIFF(s,t2.AddTime,t2.UpdateTime) asc
```
