# 站点数查询

```powershell
# 导入 IIS 管理模块
Import-Module WebAdministration

# 获取所有网站
$websites = Get-Website

# 初始化计数器
$startedCount = 0
$stoppedCount = 0

# 遍历每个网站并计数
foreach ($website in $websites) {
    if ($website.State -eq 'Started') {
        $startedCount++
    } else {
        $stoppedCount++
    }
}

# 输出结果
Write-Output "已打开的 IIS 站点数: $startedCount"
Write-Output "已关闭的 IIS 站点数: $stoppedCount"
```
