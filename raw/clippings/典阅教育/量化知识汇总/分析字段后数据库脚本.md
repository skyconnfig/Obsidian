````sql
## ✅ 最终版：金融数据仓库 - 完整 PostgreSQL 表结构脚本（支持所有年份）

```sql
-- ==================================================================
-- 🏗️ 金融数据仓库 - PostgreSQL 完整表结构脚本 (支持所有年份)
-- 基于 AKShare 数据源，支持 A股、基金、宏观、债券、期货、外汇、加密货币
-- 设计原则：模块化、高性能、可扩展、数据一致性、动态分区
-- ==================================================================

-- 设置会话参数（提升性能）
SET client_encoding = 'UTF8';
SET standard_conforming_strings = on;
SET check_function_bodies = false;
SET xmloption = content;
SET client_min_messages = warning;
SET row_security = off;

-- 创建扩展（如果尚未创建）
CREATE EXTENSION IF NOT EXISTS pg_trgm;   -- 用于文本搜索
CREATE EXTENSION IF NOT EXISTS btree_gin; -- 用于 GIN 索引的数组类型

-- ==================================================================
-- 📦 A. 基础配置表
-- ==================================================================

-- 1. 数据源配置表
CREATE TABLE data_sources (
    id SERIAL PRIMARY KEY,
    source_code VARCHAR(50) UNIQUE NOT NULL,      -- 如: akshare, tushare, wind
    source_name VARCHAR(200) NOT NULL,
    source_type VARCHAR(50) DEFAULT 'api',        -- api, web, file
    base_url VARCHAR(500),
    api_key VARCHAR(200),
    rate_limit INTEGER DEFAULT 60,                -- 每分钟最大请求次数
    status VARCHAR(20) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. 市场配置表
CREATE TABLE markets (
    id SERIAL PRIMARY KEY,
    market_code VARCHAR(20) UNIQUE NOT NULL,      -- CN, US, HK, EU
    market_name VARCHAR(100) NOT NULL,
    country VARCHAR(50),
    timezone VARCHAR(50),
    currency VARCHAR(10),
    trading_hours JSONB,                          -- {"open": "09:30", "close": "15:00", "break": ["11:30-13:00"]}
    status VARCHAR(20) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ==================================================================
-- 📈 B. 股票数据模块
-- ==================================================================

-- 1. 股票基础信息表
CREATE TABLE stocks (
    symbol VARCHAR(20) PRIMARY KEY,               -- 如: 600519.SH
    name VARCHAR(200) NOT NULL,                   -- 贵州茅台
    market_id INTEGER REFERENCES markets(id),
    industry VARCHAR(100),
    sector VARCHAR(100),
    listing_date DATE,
    delist_date DATE,
    stock_type VARCHAR(50),                       -- A股, 港股, 科创板, 北交所
    total_shares BIGINT,                          -- 总股本（万股）
    float_shares BIGINT,                          -- 流通股本（万股）
    free_share BIGINT,                            -- 自由流通股本（万股）
    status VARCHAR(20) DEFAULT 'active',
    metadata JSONB,                               -- 扩展字段：ST标志、上市板块等
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_stocks_market ON stocks(market_id);
CREATE INDEX idx_stocks_industry ON stocks(industry);
CREATE INDEX idx_stocks_sector ON stocks(sector);
CREATE INDEX idx_stocks_status ON stocks(status);

-- 2. 股票日线行情表（按年分区，动态模板）
CREATE TABLE stocks_daily (
    id BIGSERIAL,
    symbol VARCHAR(20) NOT NULL,
    trade_date DATE NOT NULL,
    open_price DECIMAL(15,6),
    high_price DECIMAL(15,6),
    low_price DECIMAL(15,6),
    close_price DECIMAL(15,6),
    adj_close_price DECIMAL(15,6),                -- 复权收盘价（推荐用此进行回测）
    volume BIGINT,                                -- 成交量（手）
    amount DECIMAL(25,2),                         -- 成交额（元）
    turnover_rate DECIMAL(10,4),                  -- 换手率（%）
    turnover_rate_f DECIMAL(10,4),                -- 自由流通股换手率（%）
    price_change DECIMAL(15,6),                   -- 涨跌额
    change_percent DECIMAL(10,4),                 -- 涨跌幅（未复权）
    pe_ratio DECIMAL(12,4),                       -- 市盈率（PE）
    pe_ttm DECIMAL(12,4),                         -- 市盈率（TTM）
    pb_ratio DECIMAL(12,4),                       -- 市净率（PB）
    ps_ratio DECIMAL(12,4),                       -- 市销率（PS）
    ps_ttm DECIMAL(12,4),                         -- 市销率（TTM）
    dv_ratio DECIMAL(10,4),                       -- 股息率（%）
    dv_ttm DECIMAL(10,4),                         -- 股息率（TTM）
    total_mv DECIMAL(25,2),                       -- 总市值（万元）
    circ_mv DECIMAL(25,2),                        -- 流通市值（万元）
    volume_ratio DECIMAL(10,4),                   -- 量比
    pre_close DECIMAL(15,6),                      -- 昨收价（前复权）
    adj_factor DECIMAL(15,10),                    -- 复权因子（核心！用于计算hfq/qfq）
    data_source_id INTEGER REFERENCES data_sources(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (symbol, trade_date)
) PARTITION BY RANGE (trade_date);

-- 创建分区索引
CREATE INDEX idx_stocks_daily_symbol ON stocks_daily(symbol);
CREATE INDEX idx_stocks_daily_date ON stocks_daily(trade_date);
CREATE INDEX idx_stocks_daily_symbol_date ON stocks_daily(symbol, trade_date);

-- 3. 股票分钟级行情表（按月分区，动态模板）
CREATE TABLE stocks_minute (
    id BIGSERIAL,
    symbol VARCHAR(20) NOT NULL,
    datetime TIMESTAMP NOT NULL,
    open_price DECIMAL(15,6),
    high_price DECIMAL(15,6),
    low_price DECIMAL(15,6),
    close_price DECIMAL(15,6),
    volume BIGINT,
    amount DECIMAL(25,2),
    data_source_id INTEGER REFERENCES data_sources(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (symbol, datetime)
) PARTITION BY RANGE (datetime);

-- 创建分区索引
CREATE INDEX idx_stocks_minute_symbol ON stocks_minute(symbol);
CREATE INDEX idx_stocks_minute_datetime ON stocks_minute(datetime);

CREATE TABLE stocks_realtime (

    change_percent DECIMAL(10,4),
    volume BIGINT,

    bid_price DECIMAL(15,6),
    ask_price DECIMAL(15,6),
    bid_volume BIGINT,
    ask_volume BIGINT,
    high_price DECIMAL(15,6),
    low_price DECIMAL(15,6),
    open_price DECIMAL(15,6),
    prev_close DECIMAL(15,6),
    update_time TIMESTAMP,
    data_source_id INTEGER REFERENCES data_sources(id),

    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ==================================================================
-- 🔧 M. 动态分区管理函数（关键！）
-- ==================================================================

-- 创建一个函数，用于自动为指定的时序表创建下一年度的分区
CREATE OR REPLACE FUNCTION create_next_year_partition(
    table_name TEXT,
    date_column TEXT,
    year_offset INTEGER DEFAULT 1
)
RETURNS VOID AS $$
DECLARE
    next_year INT := EXTRACT(YEAR FROM CURRENT_DATE) + year_offset;
    partition_name TEXT;
    start_date DATE;
    end_date DATE;
BEGIN
    start_date := make_date(next_year, 1, 1);
    end_date := make_date(next_year + 1, 1, 1);

    partition_name := table_name || '_' || next_year;

    -- 检查分区是否已存在
    IF NOT EXISTS (
        SELECT 1 FROM pg_tables WHERE tablename = partition_name AND schemaname = 'public'
    ) THEN
        EXECUTE format('CREATE TABLE %I PARTITION OF %I FOR VALUES FROM (%L) TO (%L)',
                       partition_name, table_name, start_date, end_date);
        RAISE NOTICE '✅ 已为表 %I 创建分区: %I', table_name, partition_name;
    ELSE
        RAISE NOTICE 'ℹ️ 分区 %I 已存在，无需创建。', partition_name;
    END IF;
END;
$$ LANGUAGE plpgsql;

-- SELECT create_next_year_partition('fund_nav', 'nav_date');

-- ==================================================================
-- 🧹 N. 数据库清空与重建脚本（慎用！）
-- ==================================================================

DO $$
DECLARE
    r RECORD;
BEGIN
    -- 删除所有用户表
    FOR r IN (SELECT tablename FROM pg_tables WHERE schemaname = 'public')
    LOOP
        EXECUTE 'DROP TABLE IF EXISTS ' || quote_ident(r.tablename) || ' CASCADE';
    END LOOP;

    -- 删除所有序列
    FOR r IN (SELECT sequence_name FROM information_schema.sequences WHERE sequence_schema = 'public')
    LOOP
        EXECUTE 'DROP SEQUENCE IF EXISTS ' || quote_ident(r.sequence_name) || ' CASCADE';
    END LOOP;

    -- 删除所有视图
    FOR r IN (SELECT viewname FROM pg_views WHERE schemaname = 'public')
    LOOP
        EXECUTE 'DROP VIEW IF EXISTS ' || quote_ident(r.viewname) || ' CASCADE';
    END LOOP;
END $$;
````

## ✅ 使用说明

### 🚀 如何创建新的年份分区？

1. **手动创建**：在需要时，直接运行 `create_next_year_partition` 函数。
   ```sql
   -- 为股票日线表创建2026年的分区
   SELECT create_next_year_partition('stocks_daily', 'trade_date', 2);

   -- 为基金净值表创建2026年的分区
   SELECT create_next_year_partition('fund_nav', 'nav_date', 2);
   ```
2. **自动化创建**：将上述函数调用写入一个定期任务（如每月1号的 cron 作业），自动为下一年创建分区。
