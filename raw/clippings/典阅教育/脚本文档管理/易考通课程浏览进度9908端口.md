# 易考通课程浏览进度9908端口

select CurriculumID,CurriculumName,SchoolName,ClassName,StudentNo,Name,(Round(CONVERT(float,liang),2)*100)  as '�������' from (



select CurriculumID,CurriculumName,SchoolName,ClassName,StudentNo,Name,  
(select COUNT(1) from tb_Resources q where q.CurriculumID=t.CurriculumID) as resultcount,  
(CONVERT(float,yliulan)/CONVERT(float,(select COUNT(1) from tb_Resources q where q.CurriculumID=t.CurriculumID)))  as liang  
from (  
select CurriculumID,CurriculumName,n.SchoolName,ClassName,StudentNo,b.Name,  
COUNT(1) yliulan from tb_ResourceRecord  a  
inner join dbo.tb_Student b on a.UserId=b.UserId  
inner join dbo.tb_Class c on c.C_ID=b.ClassId  
inner join tb_School n on n.S_ID=b.SchoolId  
inner join dbo.tb_Curriculum m on m.ID=a.CurriculumID  
where StudentNo between 'St00269' and 'St00418'  
group by CurriculumID,CurriculumName,StudentNo,Name,ClassName,SchoolName) t  
)y  order by liang desc,CurriculumID
