# 易考通课程浏览进度

```sql
select CurriculumID,CurriculumName,Name,TeamName,UserNo,UserName,(Round(CONVERT(float,liang),2)*100)  as '浏览进度' from (

select CurriculumID,CurriculumName,UserNo,UserName,TeamName,Name,yliulan,
(select COUNT(1) from tb_Resources q where q.CurriculumID=t.CurriculumID) as resultcount,
(CONVERT(float,yliulan)/CONVERT(float,(select COUNT(1) from tb_Resources q where q.CurriculumID=t.CurriculumID)))  as liang
 from (
select CurriculumID,CurriculumName,UserNo,b.UserName,TeamName,n.Name,COUNT(1) yliulan from tb_ResourceRecord  a
inner join tb_UserInfo b on a.UserId=b.UId
inner join tb_Team c on c.Id=b.UserClassId
inner join tb_School n on n.Id=c.SchoolId
inner join tb_Curriculum m on m.ID=a.CurriculumID
where b.AddOperator=2527

group by CurriculumID,CurriculumName,UserNo,UserName,TeamName,Name) t
)y  order by liang desc,CurriculumID

```
