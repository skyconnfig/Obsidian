def check_market_condition(context):
    """检查市场环境（增强版）"""
    if not g.USE_MARKET_FILTER:
        return True
    
    yesterday = context.previous_date
    
    # 获取指数数据
    index_data = get_price(g.MARKET_INDEX, end_date=yesterday, frequency='daily', 
                          fields=['close'], count=20, skip_paused=True)
    
    if len(index_data) < 20:
        return False
    
    # 计算均线
    ma5 = index_data['close'][-5:].mean()
    ma10 = index_data['close'][-10:].mean()
    ma20 = index_data['close'][-20:].mean()
    
    # 趋势判断
    trend_up = ma5 > ma10 > ma20
    trend_down = ma5 < ma10 < ma20
    
    # 波动率判断
    volatility = index_data['close'].pct_change().std() * np.sqrt(252)
    high_volatility = volatility > 0.25
    
    # 市场环境评分
    score = 0
    if trend_up: score += 2
    if ma5 > ma10: score += 1
    if not high_volatility: score += 1
    
    g.market_score = score
    
    # 只在趋势向上或震荡时交易
    if score < 2:
        log.info(f'市场环境不佳（评分{score}/4），暂停交易')
        return False
    
    return True

def calculate_atr(stock_code, date_obj, period=14):
    """计算ATR（平均真实波幅）"""
    try:
        data = get_price(stock_code, end_date=date_obj, frequency='daily',
                        fields=['high', 'low', 'close'], count=period+1, skip_paused=True)
        
        if len(data) < period:
            return None
        
        high = data['high']
        low = data['low']
        close = data['close']
        
        tr1 = high - low
        tr2 = abs(high - close.shift(1))
        tr3 = abs(low - close.shift(1))
        
        tr = pd.concat([tr1, tr2, tr3], axis=1).max(axis=1)
        atr = tr.rolling(window=period).mean().iloc[-1]
        
        return atr
    except:
        return None

def calculate_adaptive_macd(stock_code, date_obj):
    """自适应MACD计算"""
    try:
        # 获取历史数据计算波动率
        prices = get_price(stock_code, end_date=date_obj, frequency='daily',
                          fields=['close'], count=60, skip_paused=True)
        
        if len(prices) < 30:
            return None, None, None
        
        volatility = prices['close'].pct_change().std() * np.sqrt(252)
        
        # 根据波动率调整MACD参数
        if g.USE_ADAPTIVE_MACD:
            if volatility > 0.3:  # 高波动
                short, long, mid = 5, 12, 4
            elif volatility < 0.15:  # 低波动
                short, long, mid = 12, 26, 9
            else:  # 正常波动
                short, long, mid = g.MACD_SHORT, g.MACD_LONG, g.MACD_MID
        else:
            short, long, mid = g.MACD_SHORT, g.MACD_LONG, g.MACD_MID
        
        dif, dea, hist = MACD(stock_code, check_date=date_obj.strftime('%Y-%m-%d'),
                             SHORT=short, LONG=long, MID=mid)
        
        return dif[stock_code], dea[stock_code], hist[stock_code]
    except:
        return None, None, None

def calculate_signal_strength(stock_code, date_obj):
    """计算信号强度（多因子）"""
    try:
        score = 0
        
        # 1. MACD因子
        dif, dea, hist = calculate_adaptive_macd(stock_code, date_obj)
        if hist and hist > 0:
            score += 0.3
            if dif and dea and dif > dea:
                score += 0.2
        
        # 2. 价格位置因子
        prices = get_price(stock_code, end_date=date_obj, frequency='daily',
                          fields=['close'], count=20, skip_paused=True)
        if len(prices) >= 20:
            ma20 = prices['close'].mean()
            current = prices['close'].iloc[-1]
            if current > ma20:
                score += 0.15
        
        # 3. 量能因子
        volume_data = get_price(stock_code, end_date=date_obj, frequency='daily',
                               fields=['volume'], count=5, skip_paused=True)
        if len(volume_data) >= 5:
            avg_vol = volume_data['volume'].mean()
            if avg_vol > 0:
                score += 0.15
        
        # 4. 趋势因子
        if len(prices) >= 5:
            returns = prices['close'].pct_change().dropna()
            if len(returns) >= 3:
                up_days = sum(1 for r in returns[-3:] if r > 0)
                score += up_days * 0.067
        
        return min(score, 1.0)
    except:
        return 0

    candidates = []

            continue

            continue

        if (context.current_dt.date() - item.start_date).days < 180:
            continue

            prices = attribute_history(stock_code, 40, '1d', fields=['close'], df=False)['close']
        except:
            continue
            
        if len(prices) < 20:
            continue
        
        lastPrice = prices[-1]
        if lastPrice > g.MAX_STOCK_PRICE:
            continue
        
        # 涨幅检查
        if len(prices) < 2:
            continue
        prevPrice = prices[-2]
        increase_rate = (lastPrice - prevPrice) / prevPrice
        
        # 多因子评分
        signal_strength = calculate_signal_strength(stock_code, date_obj)
        
        if signal_strength < g.SIGNAL_STRENGTH_THRESHOLD:
            continue
        
        # MACD检查
        dif, dea, hist = calculate_adaptive_macd(stock_code, date_obj)
        if None in [dif, dea, hist]:
            continue
        
        if hist <= 0 or dif <= dea:
            continue
        
        # 综合评分
        score = signal_strength * 100
        
        # 成交量检查
        try:
            volume_data = attribute_history(stock_code, 1, '1d', fields=['money'], df=False)

                    'code': stock_code,
                    'score': score,
                    'signal_strength': signal_strength,
                    'macd_hist': hist
                })

            continue

def get_dynamic_position_size(context, stock_code, signal_strength):
    """动态仓位计算（v3.0优化版）"""
    base_position = g.POSITION_MEDIUM
    
    # 根据信号强度
    if signal_strength >= 0.8:
        base_position = g.POSITION_STRONG
    elif signal_strength >= 0.6:
        base_position = g.POSITION_MEDIUM
    else:
        base_position = g.POSITION_WEAK
    
    # 根据市场环境调整
    if hasattr(g, 'market_score'):
        if g.market_score >= 3:
            base_position *= 1.2
        elif g.market_score <= 2:
            base_position *= 0.8
    
    # 根据波动率调整
    atr = calculate_atr(stock_code, context.previous_date)
    if atr:
        try:
            current_price = attribute_history(stock_code, 1, '1d', fields=['close'], df=False)['close'][-1]
            atr_pct = atr / current_price
            
            if atr_pct > 0.03:  # 高波动
                base_position *= 0.8
            elif atr_pct < 0.015:  # 低波动
                base_position *= 1.1
        except:
            pass
    
    # 限制范围
    return max(g.POSITION_MIN, min(g.POSITION_MAX, base_position))

def stop_sell(context):
    """止损卖出（v3.0 ATR优化版）"""
    for stock_code in list(context.portfolio.positions.keys()):
        position = context.portfolio.positions[stock_code]
        
        # 计算当前价格
        try:
            y_price = attribute_history(stock_code, 1, '1d', fields=['close'], df=False)['close']
            cur_price = stock_current_price(stock_code)
        except:
            continue
        
        if cur_price is None or len(y_price) == 0:
            continue
        
        cost = position.avg_cost
        if cost == 0:
            continue
        
        # ATR动态止损
        atr = calculate_atr(stock_code, context.previous_date)
        if atr:
            atr_stop = cost - 2 * atr  # 2倍ATR止损
            if cur_price < atr_stop:
                order_target(stock_code, 0)
                g.today_sell_list.append(stock_code)
                log.info(f'ATR止损卖出: {stock_code} 价格{cur_price:.2f}')
                g.consecutive_losses += 1
                continue
        
        # 固定比例止损
        percent = (cur_price - cost) / cost
        if percent < -g.STOP_LOSS_RATE:
            order_target(stock_code, 0)
            g.today_sell_list.append(stock_code)
            log.info(f'固定止损卖出: {stock_code} 亏损{percent*100:.1f}%')
            g.consecutive_losses += 1

def sell_every_day(context):
    """日常卖出（v3.0优化版）"""
    for stock_code in list(context.portfolio.positions.keys()):
        position = context.portfolio.positions[stock_code]
        
        # 持仓天数检查
        hold_days = g.stock_hold_days.get(stock_code, 0)
        if hold_days >= g.MAX_HOLD_DAYS:
            order_target(stock_code, 0)
            g.today_sell_list.append(stock_code)
            log.info(f'到期卖出: {stock_code} 持仓{hold_days}天')
            continue
        
        cost = position.avg_cost
        cur_price = stock_current_price(stock_code)
        if cur_price is None or cost == 0:
            continue
        
        profit_pct = (cur_price - cost) / cost
        
        # 固定止盈
        if profit_pct >= g.TAKE_PROFIT_RATE:
            order_target(stock_code, 0)
            g.today_sell_list.append(stock_code)
            log.info(f'止盈卖出: {stock_code} 盈利{profit_pct*100:.1f}%')
            g.consecutive_losses = 0
            continue
        
        # 移动止盈
        highest = getattr(position, 'highest_price', cost)
        if cur_price > highest:
            highest = cur_price
        
        if highest > cost * 1.05:  # 已有5%以上盈利
            drawdown = (highest - cur_price) / highest
            if drawdown > g.TRAILING_STOP_RATE:
                order_target(stock_code, 0)
                g.today_sell_list.append(stock_code)
                log.info(f'移动止盈卖出: {stock_code} 回撤{drawdown*100:.1f}%')
                g.consecutive_losses = 0
                continue
        
        # MACD卖出信号
        date_obj = context.current_dt.date()
        dif, dea, hist = calculate_adaptive_macd(stock_code, date_obj)
        
        if hist and hist < 0 and dif and dea and dif < dea:
            order_target(stock_code, 0)
            g.today_sell_list.append(stock_code)
            log.info(f'MACD卖出: {stock_code}')

def trade_buy(context):
    """买入函数（v3.0优化版）"""
    # 检查风控
    if g.cooldown_end_date and context.current_dt.date() < g.cooldown_end_date:
        return
    
    current_hold = list(context.portfolio.positions.keys())
    if len(current_hold) >= g.MAX_HOLD_STOCKS:
        log.info(f'当前持仓已满（{len(current_hold)}只），不再买入')
        return
    
    remaining_slots = g.MAX_HOLD_STOCKS - len(current_hold)
    cash = context.portfolio.available_cash
    
    if len(g.buy_candidates) == 0 or cash < 10000:
        return
    
    buy_list = g.buy_candidates[:remaining_slots]
    
    for stock_code in buy_list:
        if stock_code in current_hold:
            continue
        
        # 计算信号强度
        signal_strength = calculate_signal_strength(stock_code, context.previous_date)
        
        # 动态仓位
        position_ratio = get_dynamic_position_size(context, stock_code, signal_strength)
        invest_amount = min(cash * position_ratio, context.portfolio.total_value * g.POSITION_MAX)
        
        if invest_amount < 5000:
            continue
        
        # 买入
        order_value(stock_code, invest_amount)
        g.stock_hold_days[stock_code] = 0
        log.info(f'买入: {stock_code} 金额{invest_amount:.0f} 仓位{position_ratio*100:.0f}% 信号{signal_strength:.2f}')
        
        cash -= invest_amount
        if cash < 10000:
            break
    
    g.today_sell_list = []

def close_check(context):
    """收盘前检查"""
    # 检查当日盈亏
    if g.prev_portfolio_value and g.prev_portfolio_value > 0:
        # 计算当日盈亏百分比
        today_pnl_pct = (context.portfolio.total_value - g.prev_portfolio_value) / g.prev_portfolio_value
        
        if today_pnl_pct < -g.MAX_DAILY_LOSS:
            log.warning(f'单日亏损超限: {today_pnl_pct*100:.2f}%，明日暂停交易')
            g.cooldown_end_date = context.current_dt.date() + timedelta(days=1)

def after_trading_end(context):
    """盘后处理"""
    # 计算并记录当日盈亏百分比
    if g.prev_portfolio_value and g.prev_portfolio_value > 0:
        today_pnl_pct = (context.portfolio.total_value - g.prev_portfolio_value) / g.prev_portfolio_value
        g.daily_pnl.append(today_pnl_pct)
        
        # 检查连续亏损
        if today_pnl_pct < 0:
            g.consecutive_losses += 1
            log.info(f'当日亏损: {today_pnl_pct*100:.2f}%，连续亏损次数: {g.consecutive_losses}')
        else:
            g.consecutive_losses = 0
            log.info(f'当日盈利: {today_pnl_pct*100:.2f}%')
    
    # 检查周亏损（过去5个交易日）
    if len(g.daily_pnl) >= 5:
        weekly_pnl = sum(g.daily_pnl[-5:])
        if weekly_pnl < -g.MAX_WEEKLY_LOSS:
            log.warning(f'单周亏损超限: {weekly_pnl*100:.2f}%，进入冷却期')
            g.cooldown_end_date = context.current_dt.date() + timedelta(days=g.COOLDOWN_DAYS)
    
    # 检查月亏损（过去20个交易日）
    if len(g.daily_pnl) >= 20:
        monthly_pnl = sum(g.daily_pnl[-20:])
        if monthly_pnl < -g.MAX_MONTHLY_LOSS:
            log.warning(f'单月亏损超限: {monthly_pnl*100:.2f}%，进入冷却期')
            g.cooldown_end_date = context.current_dt.date() + timedelta(days=g.COOLDOWN_DAYS)
    
    # 清理数据
    if len(g.daily_pnl) > 60:
        g.daily_pnl = g.daily_pnl[-60:]
    
    # 更新最后交易日
    g.last_trade_date = context.current_dt.date()

def stock_current_price(stock_code):
    """获取股票当前价格"""
    try:
        bars = attribute_history(stock_code, 1, '1d', fields=['close'], df=False)
        return bars['close'][-1] if len(bars) > 0 else None
    except:
        try:
            # 备选方法
            current_data = get_current_data()
            if stock_code in current_data:
