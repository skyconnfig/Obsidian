function scrapeShopData() {
  // 定义结果数组
  const shops = [];
  
  // 获取所有带有店铺信息的容器
  const shopContainers = document.querySelectorAll('.marginB_0.firstCell');
  
  shopContainers.forEach(container => {
    const shop = {};
    
    // 1. 店铺名称
    const nameElement = container.querySelector('.fontColorRed.cursorP.text-overflow');
    shop.name = nameElement ? nameElement.textContent.trim() : '';
    
    // 获取所有段落文本
    const paragraphs = container.querySelectorAll('p');
    paragraphs.forEach(p => {
      const text = p.textContent.trim();
      
      // 2. 店铺ID
      if (text.includes('店铺ID:')) {
        shop.id = text.replace('店铺ID:', '').trim();
      }
      // 3. 店铺地址
      else if (text.includes('店铺地址:')) {
        shop.address = text.replace('店铺地址:', '').trim();
      }
      // 4. 店铺类型
      else if (text.includes('店铺类型:')) {
        shop.type = text.replace('店铺类型:', '').trim();
      }
    });
    
    // 5. 获取所在行的其他数据
    const row = container.closest('tr');
    if (row) {
      const cells = row.querySelectorAll('td');
      
      // 查找店铺Logo (通常在第二个单元格)
      if (cells.length > 1) {
        const logoImg = cells[1].querySelector('img');
        shop.logo = logoImg ? logoImg.src : '';
      }
      
      // 获取其他数据列
      Array.from(cells).forEach((cell, index) => {
        // 跳过第一个单元格(店铺信息)和第二个单元格(Logo)
        if (index <= 1) return;
        
        const cellText = cell.textContent.trim();
        
        // 根据索引位置和内容判断数据类型
        if (index === 2) {
          // 6. 开店时间
          shop.openTime = cellText;
        } else if (index === 3) {
          // 7. 产品数量
          shop.productCount = cellText;
        } else if (index === 4) {
          // 8. 有销量产品数
          shop.soldProductCount = cellText;
        } else if (index === 5) {
          // 9. 近一日销量(pcs)
          shop.dailySales = cellText;
        } else if (index === 6) {
          // 10. 近一周销量(pcs)
          shop.weeklySales = cellText;
        } else if (index === 7) {
          // 11. 近30日销量(pcs)
          shop.monthlySales = cellText;
        } else if (index === 8) {
          // 12. 近一日/周销售额(NT$)
          shop.salesRevenue = cellText;
        } else if (index === 9) {
          // 13. 动销率
          shop.salesRate = cellText;
        } else if (index === 10) {
          // 14. 店铺评分
          shop.rating = cellText;
        } else if (index === 11) {
          // 15. 粉丝数
          shop.fans = cellText;
        } else if (index === 12) {
          // 16. 关注中
          shop.following = cellText;
        } else if (index === 13) {
          // 17. 类目销量占比
          shop.categorySalesRatio = cellText;
        }
      });
    }
    
    shops.push(shop);
  });
  
  return shops;
}

// 执行抓取并返回结果
