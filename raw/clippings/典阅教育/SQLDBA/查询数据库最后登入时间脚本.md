# 查询数据库最后登入时间脚本

demon

```plsql
declare @db varchar(50),@table varchar(50), @columnName varchar(50), @type int, @sqlstr nvarchar(max), @orderbyName varchar(10)

set @db='MobileBanking';--数据库名
set @table='P2P_UserInfo';--表明
set @columnName='UI_UserAddTime';--字段名
set @type=1;
set @orderbyName='UI_UID';--排序字段名
set @sqlstr=' use '+@db+'; select top 1 '+@columnName+' from '+@table+' order by '+@columnName ;--拼接sql语句

if(@type=1)
begin
	set @sqlstr+=' desc';
end
print @sqlstr;--打印sql语句
EXECUTE(@sqlstr);

```
