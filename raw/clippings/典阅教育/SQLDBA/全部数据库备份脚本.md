# 全部数据库备份脚本

```sql
-- 设置备份路径
DECLARE @backupPath NVARCHAR(256)
SET @backupPath = 'C:\20241108内蒙古\正式账号测试流程后20241110bak\'
 
 
-- 获取所有数据库名称
DECLARE @dbname NVARCHAR(128)
DECLARE db_cursor CURSOR FOR
SELECT name FROM sys.databases WHERE database_id > 4
 
-- 开启游标
OPEN db_cursor
 
-- 循环备份每个数据库
FETCH NEXT FROM db_cursor INTO @dbname
WHILE @@FETCH_STATUS = 0
BEGIN
    DECLARE @sql NVARCHAR(MAX)
    SET @sql = N'BACKUP DATABASE [' + @dbname + '] TO DISK = ''' + @backupPath + @dbname + '.bak' + ''' WITH NOFORMAT, NOINIT, NAME = N''' + @dbname + '-Full Database Backup'', SKIP, NOREWIND, NOUNLOAD, STATS = 10'
    
    EXEC sp_executesql @sql
    
    FETCH NEXT FROM db_cursor INTO @dbname
END
 
-- 关闭游标
CLOSE db_cursor
DEALLOCATE db_cursor
```

```sql
DECLARE @BackupPath NVARCHAR(256);
SET @BackupPath = 'C:\Backup\';  -- 请确保这个路径存在并且 SQL Server 服务账户有写入权限

DECLARE @DatabaseName NVARCHAR(128);
DECLARE @BackupFileName NVARCHAR(256);
DECLARE @SQL NVARCHAR(MAX);

-- 游标遍历所有数据库（排除系统数据库）
DECLARE db_cursor CURSOR FOR
SELECT name
FROM sys.databases
WHERE state_desc = 'ONLINE' 
  AND name NOT IN ('master', 'tempdb', 'model', 'msdb');  -- 排除系统数据库

OPEN db_cursor;
FETCH NEXT FROM db_cursor INTO @DatabaseName;

-- 循环遍历所有数据库并备份
WHILE @@FETCH_STATUS = 0
BEGIN
    -- 设置备份文件的路径和名称
    SET @BackupFileName = @BackupPath + @DatabaseName + '_' + CONVERT(NVARCHAR, GETDATE(), 112) + '.bak';

    -- 构建备份数据库的 T-SQL 语句
    SET @SQL = 'BACKUP DATABASE [' + @DatabaseName + '] TO DISK = ''' + @BackupFileName + ''' WITH INIT, FORMAT;';
    
    -- 执行备份命令
    EXEC sp_executesql @SQL;

    -- 获取下一个数据库
    FETCH NEXT FROM db_cursor INTO @DatabaseName;
END

-- 关闭和释放游标
CLOSE db_cursor;
DEALLOCATE db_cursor;

```
