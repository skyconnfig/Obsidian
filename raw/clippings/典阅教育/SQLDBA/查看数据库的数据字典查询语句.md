# 查看数据库的数据字典查询语句

```sql
-- 表信息
SELECT 
    t.name AS '表名',
    ep.value AS '表说明'
FROM 
    sys.tables t
LEFT JOIN 
    sys.extended_properties ep ON ep.major_id = t.object_id AND ep.minor_id = 0 AND ep.class = 1
ORDER BY 
    t.name;

-- 列信息
SELECT 
    t.name AS '表名',
    c.name AS '列名',
    ep.value AS '列说明',
    st.name AS '数据类型',
    CASE 
        WHEN st.name IN ('char', 'varchar', 'nchar', 'nvarchar') THEN c.max_length 
        WHEN st.name IN ('numeric', 'decimal') THEN c.precision 
        ELSE NULL 
    END AS '数值精度',
    CASE 
        WHEN st.name IN ('numeric', 'decimal') THEN c.scale 
        ELSE NULL 
    END AS '数值刻度',
    CASE WHEN c.is_nullable = 1 THEN '是' ELSE '否' END AS '可空',
    CASE WHEN c.is_identity = 1 THEN '是' ELSE '否' END AS '标识列'
FROM 
    sys.tables t
INNER JOIN 
    sys.columns c ON t.object_id = c.object_id
INNER JOIN 
    sys.types st ON c.user_type_id = st.user_type_id
LEFT JOIN 
    sys.extended_properties ep ON ep.major_id = t.object_id AND ep.minor_id = c.column_id AND ep.class = 1
ORDER BY 
    t.name, c.column_id;

-- 主键信息
SELECT 
    t.name AS '表名',
    kc.name AS '主键名称',
    cols.name AS '列名'
FROM 
    sys.key_constraints kc
INNER JOIN 
    sys.tables t ON kc.parent_object_id = t.object_id
INNER JOIN 
    sys.index_columns ic ON kc.parent_object_id = ic.object_id AND kc.unique_index_id = ic.index_id
INNER JOIN 
    sys.columns cols ON ic.object_id = cols.object_id AND ic.column_id = cols.column_id
WHERE 
    kc.type = 'PK'
ORDER BY 
    t.name, kc.name;

-- 外键信息
SELECT 
    fk.name AS '外键名称',
    OBJECT_NAME(fk.parent_object_id) AS '父表',
    c1.name AS '父表列',
    OBJECT_NAME(fk.referenced_object_id) AS '引用表',
    c2.name AS '引用表列'
FROM 
    sys.foreign_keys fk
INNER JOIN 
    sys.foreign_key_columns fkc ON fk.object_id = fkc.constraint_object_id
INNER JOIN 
    sys.columns c1 ON fkc.parent_object_id = c1.object_id AND fkc.parent_column_id = c1.column_id
INNER JOIN 
    sys.columns c2 ON fkc.referenced_object_id = c2.object_id AND fkc.referenced_column_id = c2.column_id;

-- 索引信息
SELECT 
    t.name AS '表名',
    ind.name AS '索引名称',
    COL_NAME(ic.object_id, ic.column_id) AS '列名',
    CASE WHEN ind.is_unique = 1 THEN '是' ELSE '否' END AS '唯一索引'
FROM 
    sys.indexes ind
INNER JOIN 
    sys.index_columns ic ON ind.object_id = ic.object_id AND ind.index_id = ic.index_id
INNER JOIN 
    sys.tables t ON ind.object_id = t.object_id
WHERE 
    ind.is_primary_key = 0 AND ind.type_desc <> 'HEAP'
ORDER BY 
    t.name, ind.name;

```
